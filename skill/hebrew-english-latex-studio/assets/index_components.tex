% Do not change directional folios. A separate expandable locator is deferred
% through protected@write and expands only when its page/float ships out.
\newcommand{\hllpdf@indexrawpage}{\arabic{page}}
\newcommand{\hllpdf@indexcheckformat}{%
  \edef\hllpdf@indexexpectedpage{\noexpand\foreignlanguage{english}{\noexpand\arabic{page}}}%
  \def\hllpdf@indexresetpage{\foreignlanguage{english}{\csname @arabic\endcsname\c@page}}%
  \ifx\thepage\hllpdf@indexexpectedpage\else
    \ifx\thepage\hllpdf@indexresetpage\else
      \PackageError{hllpdf-index}{Index requires the core Arabic folio contract}{Retain the core Arabic formatter; custom page numbering is unsupported.}
    \fi
  \fi}
\newcommand{\hllpdf@indexwritecontext}{%
  \hllpdf@indexcheckformat
  \let\hllpdf@indexoriginalwrite\protected@write
  \long\def\protected@write##1##2##3{%
    \hllpdf@indexoriginalwrite{##1}{\let\hllpdf@indexrawpage\relax ##2}{##3}}}
\pretocmd{\imki@wrindexentrysplit}{\hllpdf@indexwritecontext}{}%
  {\PackageError{hllpdf-index}{Index writer contract changed}{The supported writer hook could not be installed.}}
\patchcmd{\@wrindex}{\thepage}{\hllpdf@indexrawpage}{}%
  {\PackageError{hllpdf-index}{Index page contract changed}{The supported page hook could not be installed.}}

% All index text defaults to the normal body face, size and black color.
% Babel's logical raggedright aligns to the locale's physical starting edge.
% This backend mirrors hangindent against the page base, so opposite-locale
% paragraphs need the opposite sign to indent their own physical leading edge.
\newcommand{\LakitIndexEntry}[2]{%
  \par\iflanguage{#1}{}{\PackageError{hllpdf-index}{Generated index locale mismatch}{Retain the paired index.ist language environment.}}%
  \normalfont\normalsize\color{black}%
  \raggedright\noindent\iflanguage{\lakitmainlanguage}{\hangindent=1.25em}{\hangindent=-1.25em}\hangafter=1 #2}
\newcommand{\LakitIndexLocators}[1]{\nobreak\hspace{.65em}\babelsublr{#1}}
\newcommand{\LakitIndexEnd}{\par}
\newcommand{\LakitIndexGroup}[1]{%
  \par\addvspace{.8\baselineskip}\Needspace{3\baselineskip}%
  \ifstrequal{#1}{E}{%
    \begin{otherlanguage}{english}\raggedright\normalfont\normalsize\bfseries\noindent English\par\end{otherlanguage}}{%
    \ifstrequal{#1}{H}{%
      \begin{otherlanguage}{hebrew}\raggedright\normalfont\normalsize\bfseries\noindent עברית\par\end{otherlanguage}}{%
      \PackageError{hllpdf-index}{Unknown index group '#1'}{Only English E and Hebrew H group keys are supported.}}}}

\prop_new:N \g_hllpdf_index_displays_prop
\newcommand{\hllpdfindexadd}[4]
  {
    \regex_match:nnF {\A [0-9]{6} \Z} {#2}
      {\PackageError{hllpdf-index}{Index~sort~key~must~be~six~ASCII~digits}{Use~explicit~equal-width~ranks;~display~text~does~not~determine~collation.}}
    \exp_args:Nnx \regex_match:nnT {[@!|"]} {\tl_to_str:n {#3}}
      {\PackageError{hllpdf-index}{Reserved~makeindex~character~in~display}{Avoid~literal~at-sign,~exclamation,~pipe~or~quote;~use~a~tested~TeX~display~command.}}
    \tl_set:Nx \l_tmpb_tl {\tl_to_str:n {#3}}
    \prop_get:NnNTF \g_hllpdf_index_displays_prop {#1/#2} \l_tmpa_tl
      {\tl_if_eq:NNF \l_tmpa_tl \l_tmpb_tl
        {\PackageError{hllpdf-index}{Index~rank~reused~for~different~display}{Use~a~unique~rank~per~term;~range~endpoints~must~repeat~the~same~display.}}}
      {\prop_gput:NnV \g_hllpdf_index_displays_prop {#1/#2} \l_tmpb_tl}
    \str_case:nnF {#1}
      {
        {english}{\exp_args:Nx \index {\tl_to_str:n {E#2@\begin{otherlanguage}{english}\LakitIndexEntry{english}{#3}|#4hyperpage}}}
        {hebrew}{\exp_args:Nx \index {\tl_to_str:n {H#2@\begin{otherlanguage}{hebrew}\LakitIndexEntry{hebrew}{#3}|#4hyperpage}}}
      }
      {\PackageError{hllpdf-index}{Index~language~must~be~english~or~hebrew}{Declare~each~entry's~paragraph~locale~explicitly.}}
  }
