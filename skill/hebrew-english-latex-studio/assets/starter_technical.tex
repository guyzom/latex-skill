\section{השאלה והנחות העבודה}
נחשב סכום משוקלל של סדרת ערכים. נניח שהערכים והמשקלים נתונים ושמספר
המשקלים שווה למספר הערכים.
\section{הגדרה ופיתוח}
\begin{equation}\label{eq:weighted}
y=\sum_{i=1}^{n}w_i x_i.
\end{equation}
ב־\eqref{eq:weighted}, $x_i$ הוא האיבר, $w_i$ הוא המשקל ו־$n$ הוא מספר
האיברים.
\begin{exbox}[frametitle={בדיקה פשוטה}]
עבור $n=2$, אם $w_1=w_2=1$, נקבל $y=x_1+x_2$.
\end{exbox}
\section{מימוש לדוגמה}
\begin{codebox}[frametitle={\en{Python}}]
\begin{asmblock}
\begin{lstlisting}[style=pystyle]
def weighted_sum(values, weights):
    if len(values) != len(weights):
        raise ValueError("Length mismatch")
    return sum(w * x for w, x in zip(weights, values))
\end{lstlisting}
\end{asmblock}
\end{codebox}
