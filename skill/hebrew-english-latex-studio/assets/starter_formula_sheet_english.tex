\section{Arithmetic progression}
\[
S_n=\frac{n}{2}(a_1+a_n),\qquad a_n=a_1+(n-1)d.
\]
Here $n$ is the number of terms, $a_1$ the first term, and $d$ the common difference.
\section{Finite geometric progression}
\[
S_n=a_1\frac{1-q^n}{1-q},\qquad q\neq1.
\]
For $q=1$, use $S_n=n a_1$.
\section{Chain rule}
\[
\frac{d}{dx}f(g(x))=f'(g(x))\,g'(x).
\]
The functions must be differentiable at the relevant points.
