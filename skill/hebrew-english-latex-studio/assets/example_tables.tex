{\LARGE\bfseries\noindent מאגר תבניות טבלאות\par}\medskip
\noindent העתק את התבנית שאתה צריך והחלף תוכן. הטבלאות הקצרות ממורכזות ברוחב התוכן, עם כיתוב ממוספר.
בחר רוחב עמודת טקסט לפי התוכן; מסגרת \code{tablebox} היא אפשרות מפורשת. שני כללים לטבלה מעורבת: כשהתוכן עברי־ראשי,
הגדר סדר עמודות אנגלי, בודד תאים עבריים ב-\code{\textbackslash h},
והצהר את עמודת המפתח אחרונה — היא נוחתת מימין,
והקורא פוגש אותה ראשונה; כשהתוכן \en{LTR}־ראשי (קוד, נוסחאות, טבלת אמת), עטוף ב-\en{otherlanguage}.
תבניות \en{1–6} הן משפחת ה-\en{booktabs} הפתוחה (קווים אופקיים בלבד); תבניות 8, 9 ו-12
הן דוגמאות לרשת מלאה כאשר מסמך המקור מבקש זאת במפורש.
\bigskip

% +--- PATTERN 1: basic data table --------------------------------
% | USE WHEN: the default. A few columns, Hebrew headers, cells that
% |           may hold short math islands.
% | KEY:      booktabs rules (\toprule/\midrule/\bottomrule); short
% |           math as \en{$...$}; caption via \tabcap. Hebrew-primary
% |           -> explicit English table order, Hebrew cell islands, KEY column declared LAST (lands rightmost).
% +----------------------------------------------------------------
\patternheading{1. טבלת נתונים בסיסית}
\begin{center}
\begin{otherlanguage}{english}
\begin{tabular}{@{}lll@{}}
\toprule
\h{\textbf{הערה}} & \h{\textbf{סיבוכיות זמן}} & \h{\textbf{שיטה}}\\
\midrule
\h{ללא דרישת מיון.} & \en{$O(n)$}        & \h{חיפוש לינארי}\\
\h{דורש מערך ממוין.} & \en{$O(\log n)$}   & \h{חיפוש בינארי}\\
\h{תלוי בפונקציית הגיבוב.} & \h{\en{$O(1)$} בממוצע} & \h{טבלת גיבוב}\\
\bottomrule
\end{tabular}
\end{otherlanguage}
\tabcap{השוואת שיטות חיפוש.}
\end{center}
\bigskip
