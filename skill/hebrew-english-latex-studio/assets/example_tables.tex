% example_tables.tex (v3.40.2) — GALLERY + smoke test for the TABLE patterns.
% It \input{}s the REAL preamble.tex, so compiling it both (a) tests the actual
% environment and (b) gives you copy-paste templates proven to render cleanly.
%
%   lualatex -interaction=nonstopmode example_tables.tex   (or via build.sh)
%   pdftoppm -png -r 150 example_tables.pdf pg   &&   view EVERY page
%
% This is a LIBRARY, not a mandate: grab the pattern you need and swap the
% content. Content here is deliberately topic-neutral (math / physics / CS)
% because the skill serves a wide range of subjects.
%
% THE ONE DECISION that governs every mixed-direction table:
%   * Columns ALWAYS lay out LEFT-to-right — the FIRST declared column lands
%     LEFTMOST, in a plain tabular exactly as in otherlanguage (measured on
%     rendered pages; an earlier version of this comment claimed plain
%     tabular is RTL — it is not).
%   * Hebrew-primary table (labels & cells mostly Hebrew, scanned from the
%     right) -> plain tabular, and declare the KEY column LAST so it lands
%     RIGHTMOST — the RTL reader meets the key before its explanation
%     (side-by-side render comparison: key-on-the-right reads naturally,
%     key-on-the-left forces the eye to jump).
%   * LTR-primary table (code, truth tables, sequences — content that must
%     READ left-to-right) -> wrap the tabular in otherlanguage{english}:
%     it left-aligns the LTR cells uniformly and keeps the whole block in
%     LTR reading order.
%   Every table below is also wrapped in tablebox + \tabcap (uniform slate
%   frame + auto-numbered "טבלה N.M." caption inside) — never \caption*.
%
% TWO MORE MEASURED FACTS (v3.40.2, calibration renders):
%   * SIDE-BY-SIDE minipages are the OPPOSITE of columns: the minipage
%     declared FIRST lands RIGHTMOST (paragraph-level blocks follow the
%     page's RTL flow; tabular columns do not). See pattern 7.
%   * DO NOT use longtable here: under the RTL base it REVERSES the declared
%     column order relative to plain tabular (measured — first declared
%     column landed rightmost). The preamble does NOT load longtable, and with
%     it loaded by hand inside tablebox the output is broken: the first page
%     comes out empty and \tabcap never prints (re-measured on v4.7.3; the
%     older head/foot mechanism did not reproduce). Multi-page tables need a
%     dedicated verification cycle first (DEFECTS BIDI-25).
% ============================================================================

\newcommand{\patternheading}[1]{%
  \Needspace{11\baselineskip}%
  {\large\bfseries\noindent #1\par}\smallskip}
% (No \setcounter{chapter} hack needed: since v3.30 \tabcap prints "טבלה N."
%  while no \chapter exists, and "טבלה C.N" once chapters start.)

{\LARGE\bfseries\noindent מאגר תבניות טבלאות\par}\medskip
\noindent העתק את התבנית שאתה צריך והחלף תוכן. כל טבלה כאן עוברת \code{tablebox} ו-\code{tabcap}
— מסגרת אחידה וכותרת ממוספרת בפנים. שני כללים לטבלה מעורבת: כשהתוכן עברי־ראשי,
השתמש ב-\code{tabular} רגיל \emph{והצהר את עמודת המפתח אחרונה} — היא נוחתת מימין,
והקורא פוגש אותה ראשונה; כשהתוכן \en{LTR}־ראשי (קוד, נוסחאות, טבלת אמת), עטוף ב-\en{otherlanguage}.
תבניות \en{1–6} הן משפחת ה-\en{booktabs} הפתוחה (קווים אופקיים בלבד); תבניות 8, 9 ו-12
הן משפחת הרשת המלאה — המראה ה״סטנדרטי״ של מסמכי מקור (מסגרת + מפרידים אנכיים).
\bigskip

% +--- PATTERN 1: basic data table --------------------------------
% | USE WHEN: the default. A few columns, Hebrew headers, cells that
% |           may hold short math islands.
% | KEY:      booktabs rules (\toprule/\midrule/\bottomrule); short
% |           math as \en{$...$}; caption via \tabcap. Hebrew-primary
% |           -> plain tabular, KEY column declared LAST (lands rightmost).
% +----------------------------------------------------------------
\patternheading{1. טבלת נתונים בסיסית}
\begin{tablebox}\centering\small
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{הערה} & \textbf{סיבוכיות זמן} & \textbf{שיטה}\\
\midrule
ללא דרישת מיון.        & \en{$O(n)$}        & חיפוש לינארי\\
דורש מערך ממוין.       & \en{$O(\log n)$}   & חיפוש בינארי\\
תלוי בפונקציית הגיבוב. & \en{$O(1)$} בממוצע & טבלת גיבוב\\
\bottomrule
\end{tabular}
\tabcap{השוואת שיטות חיפוש.}
\end{tablebox}
\bigskip

% | USE WHEN: a column holds code or dense LTR that must left-align

% | KEY:      wrap the WHOLE tabular in otherlanguage{english};
% |           Hebrew labels then need \h{}; a multi-line code cell =

% ---- numeric columns: decimal alignment WITHOUT any package (v3.37) --------
% dcolumn and siunitx both SCRAMBLE under the RTL base (render-proven:
% D{.}{.}{3.3} shipped "12.75" as ".7512"; \SI{3.5}{eV} ships "eV 3.5") — do
% not load them. The bidi-safe recipe: (a) give every value in the column the
% SAME number of decimals, (b) declare the column `l` — under babel-RTL the
% l/r colspec letters swap their VISUAL meaning, so `l` right-aligns (measured:
% all decimal points at x=322.7 exactly). Hebrew headers need no tricks here.
\begin{tablebox}\centering\small
\begin{tabular}{l l}
\toprule
\textbf{שם} & \textbf{ערך} \\
\midrule
אלפא & 1.500 \\
בטא & 12.750 \\
גמא & 103.200 \\
דלתא & 0.008 \\
\bottomrule
\end{tabular}
\tabcap{יישור עשרוני: ספרות אחידות + עמודת \code{l} (ימין ויזואלי תחת RTL).}
\end{tablebox}
