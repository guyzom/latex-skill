% Keep native class heading numbering visible. With mathtools, only equations
% used by \eqref/\refeq receive automatic numbers. An author may explicitly
% mark a central, unreferenced equation with \noeqref{eq:central} in prose.
% Routine derivations use equation*/align* or \[...\]. Avoid invented tags.
\mathtoolsset{showonlyrefs=true,showmanualtags=false}
\lakitcaptionnumbering{continuous}
\renewcommand{\lakitcaptionpunctuation}{:}

\iflanguage{hebrew}{\renewcommand{\abstractname}{תקציר}}%
  {\renewcommand{\abstractname}{Abstract}}

% In the ordinary article title block, literal \author{} / \date{} should not
% reserve invisible metadata rows above the abstract. Retain the class's own
% sizes, nonempty author tabular (including \and), date and outer \maketitle
% handling of \thanks, footnotes and title-page policy. Metadata remains live
% until \maketitle: macros supplied as title/author/date expand at typesetting.
% Explicit titlepage articles and report titles keep their native class path.
\makeatletter
\@ifclassloaded{article}{%
  \if@titlepage\else
    \renewcommand{\@maketitle}{%
      \newpage\null\vskip 2em%
      \begin{center}%
        \normalfont\rmfamily\let\footnote\thanks
        {\LARGE\@title\par}%
        \ifx\@author\@empty\else
          \vskip 1.5em%
          {\large\lineskip .5em%
            \begin{tabular}[t]{c}\@author\end{tabular}\par}%
        \fi
        \ifx\@date\@empty\else
          \vskip 1em{\large\@date\par}%
        \fi
      \end{center}\par\vskip 1.5em}%
  \fi
}{}
\makeatother
