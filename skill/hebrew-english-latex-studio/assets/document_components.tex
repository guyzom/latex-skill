% \sourcelink{URL}{descriptive label}{why this source is here}
% Put English labels in \en{...}; the destination is not printed as raw prose.
% The role follows the surrounding language, not the link title.
% The source role is textual (not conveyed by color alone). Long labels can wrap.
% This is a SOURCE ANNOTATION, not a substitute for claim-level scientific citations.
\newcommand{\sourcelink}[3]{%
  \par\addvspace{6pt}\begingroup
  \small\noindent\textbf{\iflanguage{english}{Source:}{מקור:}}\enspace
  \href{#1}{\textcolor{cBlue}{\textbf{#2}}}\par\nobreak
  \noindent #3\par
  \endgroup\addvspace{6pt}}


% \bibentryprose{english|hebrew}{complete bibliography entry}
% Use immediately after native \bibitem. The outer bibliography retains its
% heading, list spacing, label/counter and cite target. Only this breakable
% entry paragraph changes language; no box, font reset or size reduction.
% A label pending in the outer list must stay on its original physical side
% when the paragraph direction changes. Shift its native label box
% across the live list body width, preserving its width and custom label spacing.
\makeatletter
\newcommand{\bibentryprose}[2]{%
  \par\begingroup
  \def\hllpdf@bibliographyenv{thebibliography}%
  \ifx\@currenvir\hllpdf@bibliographyenv\else
    \PackageError{hllpdf}{bibentryprose requires thebibliography}%
      {Place it immediately after the native bibitem command.}%
  \fi
  \if@inlabel\else
    \PackageError{hllpdf}{bibentryprose requires a pending bibitem label}%
      {Use one complete entry paragraph immediately after bibitem.}%
  \fi
  \edef\hllpdf@bibentrydir{\the\pardirection}%
  \ifstrequal{#1}{english}{\selectlanguage{english}}{%
    \ifstrequal{#1}{hebrew}{\selectlanguage{hebrew}}{%
      \PackageError{hllpdf}{bibentryprose language '#1' must be english or hebrew}%
        {Choose the actual language of this complete bibliography entry.}%
    }%
  }%
  \@tempdima=\wd\@labels
  \ifnum\pardirection=\hllpdf@bibentrydir\relax\else
    \@tempdima=\wd\@labels
    \ifnum\pardirection=0
      \global\setbox\@labels\hbox dir TLT to\@tempdima{\kern\linewidth\box\@labels\hss}%
    \else
      \global\setbox\@labels\hbox dir TLT to\@tempdima{\kern-\linewidth\box\@labels\hss}%
    \fi
  \fi
  #2\par
  \endgroup}
\makeatother

% \glossaryentry[definition language]{term}{definition}: a local paragraph.
% An omitted locale inherits the surrounding paragraph; english/hebrew are
% explicit choices. Minority-language terms still require \en or \h.
% Logical raggedright follows Babel's active reading direction: physically
% left in English and right in Hebrew. Its scope ends with this paragraph.
% Keep the term, dash and first definition word connected without fixed columns.
% Sentence punctuation remains authored; a signed quantity belongs in math.
\newcommand{\glossaryentry}[3][]{%
  \par\begingroup
  \ifblank{#1}{}{%
    \ifstrequal{#1}{english}{\selectlanguage{english}}{%
      \ifstrequal{#1}{hebrew}{\selectlanguage{hebrew}}{%
        \PackageError{hllpdf}{glossaryentry language '#1' must be english or hebrew}%
          {Omit the optional argument to inherit the local paragraph language.}%
      }%
    }%
  }%
  \raggedright%
  \noindent #2\nobreak\hspace{.35em}—\nobreak\hspace{.35em}#3\par
  \endgroup}

% Two optional atomic modes: framedtable fits content; semanticframedtable
% matches the active semantic-box geometry. Neither chooses a table schema.
% [short list caption]{column count}{column specification}{caption}{label}
% Natural l/r/c columns stay natural. Author p/m widths for long prose; the
% local \framedtablewidth budget deducts frame, cell and intercolumn padding.
% Unlike flowtable this environment deliberately does not span pages.
\makeatletter
\newlength{\hllpdf@framecolumns}
\newcommand{\framedtablewidth}{\hllpdf@framecolumns}
\newsavebox{\hllpdf@framedrows}
\newsavebox{\hllpdf@framedbody}
\newsavebox{\hllpdf@framedcaption}
\newlength{\hllpdf@frameinner}
\newlength{\hllpdf@frameouter}
\newlength{\hllpdf@frameparent}
\newlength{\hllpdf@framesurround}
\newlength{\hllpdf@framevertical}
\newcommand{\hllpdf@frametable}[7]{%
  \par\begingroup
  \ifnum\numexpr#3\relax<1
    \PackageError{hllpdf}{framedtable requires a positive column count}%
      {Choose the actual number of columns from the content, with no fixed schema.}%
  \fi
  \setlength{\hllpdf@frameparent}{\linewidth}%
  \ifnum#1=0
    \mdfsetup{style=\hllpdftableframestyle}%
    \def\hllpdf@framealign{center}%
  \else
    \mdfsetup{style=\hllpdfsemanticframestyle}%
    \def\hllpdf@framealign{left}%
  \fi
  % Resolve the same mdframed width equation used by real semantic/table boxes.
  % This reads live theme geometry without typesetting a dummy semantic box or
  % executing its hooks. Snapshot it before tablebox applies its own defaults.
  \mdfsetup{userdefinedwidth=\hllpdf@frameparent}%
  \mdf@styledefinition
  \mdf@horizontalmargin@equation
  \setlength{\hllpdf@frameinner}{\mdf@horizontalspaceofbox}%
  \hsize=\hllpdf@frameparent
  \setlength{\hllpdf@framesurround}{\dimexpr\hllpdf@frameparent-\hllpdf@frameinner\relax}%
  \setlength{\hllpdf@framevertical}{\dimexpr
    \mdf@innertopmargin@length+\mdf@innerbottommargin@length+
    \mdf@skipabove@length+\mdf@skipbelow@length+
    \ifmdf@topline\mdf@linewidth@length\else0pt\fi+
    \ifmdf@bottomline\mdf@linewidth@length\else0pt\fi\relax}%
  \edef\hllpdf@setframestyle{\noexpand\mdfdefinestyle{hllpdfmeasuredtable}{%
    leftmargin=\the\mdf@leftmargin@length,rightmargin=\the\mdf@rightmargin@length,
    innerleftmargin=\the\mdf@innerleftmargin@length,innerrightmargin=\the\mdf@innerrightmargin@length,
    innertopmargin=\the\mdf@innertopmargin@length,innerbottommargin=\the\mdf@innerbottommargin@length,
    skipabove=\the\mdf@skipabove@length,skipbelow=\the\mdf@skipbelow@length,
    linewidth=\the\mdf@linewidth@length,roundcorner=\the\mdf@roundcorner@length,
    leftline=\ifmdf@leftline true\else false\fi,rightline=\ifmdf@rightline true\else false\fi,
    topline=\ifmdf@topline true\else false\fi,bottomline=\ifmdf@bottomline true\else false\fi}}%
  \hllpdf@setframestyle
  \setlength{\hllpdf@framecolumns}{\dimexpr\hllpdf@frameinner-2\tabcolsep*#3\relax}%
  \sbox{\hllpdf@framedrows}{\begin{otherlanguage}{english}\begin{tabular}{#4}#7\end{tabular}\end{otherlanguage}}%
  \ifdim\wd\hllpdf@framedrows>\hllpdf@frameinner
    \PackageError{hllpdf}{framedtable exceeds the available width}%
      {Wrap prose with p/m columns within framedtablewidth; do not scale the table.}%
  \fi
  % Capture description nodes once, then unbox them into the final paragraph.
  % They remain breakable at the safe width; citations/links are not executed
  % twice. Plain label measurement has no counter or navigation side effects.
  \sbox{\hllpdf@framedcaption}{\lakitcaptionstyle #5}%
  \sbox\@tempboxa{\advance\c@tbl by 1\relax
    \lakitcaptionstyle\tablename~\en{\thetbl}\lakitcaptionpunctuation\enspace}%
  \@tempdima=\dimexpr\wd\@tempboxa+\wd\hllpdf@framedcaption\relax
  \ifnum#1=0
    \ifdim\@tempdima<\hllpdf@frameinner
      \setlength{\hllpdf@frameinner}{\@tempdima}%
    \fi
    \ifdim\wd\hllpdf@framedrows>\hllpdf@frameinner
      \setlength{\hllpdf@frameinner}{\wd\hllpdf@framedrows}%
    \fi
  \fi
  \let\hllpdf@frametitle\lakinrtitle
  \renewcommand{\lakinrtitle}[1]{\hllpdf@frametitle{#5}}%
  \begin{lrbox}{\hllpdf@framedbody}%
    \begin{minipage}{\hllpdf@frameinner}\centering
      \usebox{\hllpdf@framedrows}%
      \lakintabframedtrue
      \tabcap[#2]{\label{#6}\unhcopy\hllpdf@framedcaption}%
    \end{minipage}%
  \end{lrbox}%
  \setlength{\hllpdf@frameouter}{\dimexpr\hllpdf@frameinner+\hllpdf@framesurround\relax}%
  \ifdim\dimexpr\ht\hllpdf@framedbody+\dp\hllpdf@framedbody+\hllpdf@framevertical\relax>\textheight
    \PackageError{hllpdf}{framedtable exceeds one page}%
      {Split this atomic table or use an unframed flowtable outside a box.}%
  \fi
  % Target this frame's top, using its actual padding rather than the legacy
  % cushion. The existing close hook traverses the captured caption anchor.
  \renewcommand{\lakintabcushion}{\dimexpr\mdf@innertopmargin@length+
    \ifmdf@topline\mdf@linewidth@length\else0pt\fi\relax}%
  \begin{tablebox}[style=hllpdfmeasuredtable,userdefinedwidth=\hllpdf@frameouter,
    align=\hllpdf@framealign,nobreak=true,frametitle={},
    startinnercode=\lakintabboxopen,endinnercode=\lakintabboxclose]
    \global\lakintabcapseentrue
    \noindent\usebox{\hllpdf@framedbody}%
  \end{tablebox}%
  \endgroup
}

}{ }
\NewDocumentEnvironment{semanticframedtable}{O{} m m m m +b}{%
