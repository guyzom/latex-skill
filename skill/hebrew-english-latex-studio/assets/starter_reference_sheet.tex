\begin{referenceentry}
\section{וקטורים ומספרים מרוכבים}
לווקטורים ממשיים, $\mathbf a\cdot\mathbf b=\sum_i a_i b_i$; לווקטורים מרוכבים,
$\langle u,v\rangle=u^\dagger v$.
\[
\|v\|=\sqrt{v^\dagger v},\qquad
P_u v=u\frac{u^\dagger v}{u^\dagger u},\quad u\neq0.
\]
השארית $r=v-P_u v$ מאונכת ל־$u$: מתקיים $u^\dagger r=0$.
\subsection{הצגה מרוכבת}
עבור $z=x+\mathrm i y$, כאשר $\mathrm i^2=-1$,
\[
|z|^2=x^2+y^2,\qquad z^*=x-\mathrm i y,
\]
\[
\mathrm e^{\mathrm i\theta}=\cos\theta+\mathrm i\sin\theta.
\]
בכתיב $z=r\mathrm e^{\mathrm i\theta}$ מתקיים $r\geq0$; הפאזה מוגדרת
מודולו $2\pi$ ואינה מוגדרת כאשר $r=0$.
\end{referenceentry}

\begin{referenceentry}
\section{מטריצות וערכים עצמיים}
\[
(AB)^\dagger=B^\dagger A^\dagger,\qquad
\operatorname{tr}(AB)=\operatorname{tr}(BA).
\]
זהות העקבה דורשת ממדים מתאימים. זוג עצמי מקיים
$Av=\lambda v$, $v\neq0$. למטריצה הרמיטית,
\[
A=U\Lambda U^\dagger,\qquad U^\dagger U=I,
\]
כאשר $\Lambda$ אלכסונית וממשית. מטריצה ריבועית הפיכה אם ורק אם
$\det A\neq0$; אז $A^{-1}A=I$ ו־$\det(A^{-1})=1/\det A$.
\end{referenceentry}

\subsection{קירוב מקומי}
לפונקציה חלקה $f$ ליד $x_0$, נסמן $h=x-x_0$:
\[
f(x_0+h)=f(x_0)+f'(x_0)h+O(h^2).

\begin{referenceentry}
\section{חשבון וקטורי}
לשדה סקלרי $f$ ולשדה וקטורי $\mathbf F$,
\[
\nabla f=(\partial_x f,\partial_y f,\partial_z f),
\]
\[
\nabla\cdot\mathbf F=\partial_x F_x+\partial_y F_y+\partial_z F_z.
\]
הגרדיאנט מתאר עלייה מהירה ביותר; הדיברגנץ מודד זרימה מקומית החוצה.
לשדה בעל סדירות מספקת בתחום חסום,
\[
\int_V\nabla\cdot\mathbf F\,\mathrm dV
=\oint_{\partial V}\mathbf F\cdot\mathbf n\,\mathrm dS,
\]
כאשר $\mathbf n$ הוא הנורמל היחידה החיצוני.
\end{referenceentry}

\begin{referenceentry}
\section{הסתברות ואי־ודאות}
לערכים בדידים $x_i$ ולהסתברויות $p_i$,
\[
p_i\geq0,\quad\sum_i p_i=1,\quad\mu=\sum_i p_i x_i,
\]
\[
\sigma^2=\sum_i p_i(x_i-\mu)^2.
\]
עבור $X,Y$ ממשיים בעלי שונויות סופיות ו־$a,b$ ממשיים,
\[
\begin{aligned}
\operatorname{Var}(aX+bY)
={}&a^2\operatorname{Var}(X)\\
 &+b^2\operatorname{Var}(Y)\\
 &+2ab\operatorname{Cov}(X,Y).
\end{aligned}
\]
אי־תלות גוררת $\operatorname{Cov}(X,Y)=0$. כאשר $a=b=1$, שונויות
של משתנים בלתי תלויים מתחברות.
\end{referenceentry}

\begin{referenceentry}
\section{מצב קוונטי ומדידה}
מצב טהור מייחס אמפליטודות מרוכבות לבסיס אורתונורמלי דו־מצבי נתון.
\begin{referenceformula}
\begin{equation}\label{eq:qubit}
\ket{\psi}=\alpha\ket0+\beta\ket1,
\end{equation}
\[
|\alpha|^2+|\beta|^2=1.
\]
פאזה כוללת משותפת אינה ניתנת לצפייה.
\end{referenceformula}
מדידת~\eqref{eq:qubit} נותנת תוצאות $0$ ו־$1$ בהסתברויות
$|\alpha|^2$ ו־$|\beta|^2$, בהתאמה.
\subsection{אופרטורים ותוחלת}
לאופרטור הרמיטי $A$ ולמצב מנורמל $\ket\psi$,
\[
\langle A\rangle=\bra\psi A\ket\psi,
\]
\[
(\Delta A)^2=\langle A^2\rangle-\langle A\rangle^2.
\]
במצב עצמי של $A$ מתקיים $(\Delta A)^2=0$.
\end{referenceentry}
